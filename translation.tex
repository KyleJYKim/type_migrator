\section{Formalized TypeSpec and Elixir Type}

\begin{figure} [H]
   \begin{grammarfig}
      \textbf{Annotations} & \FWA &\bnfeq& 
      \TSAn \bnfor \varepsilon
      \\
      \textbf{Expressions} & \TSE &\bnfeq& 
      \TSFn{\TST} \bnfor 
      \TSFn{\TST}\,\,\texttt{when}\,\,\ov{\texttt{$\alpha$:}\,\TST}
      \\[2mm]
      \textbf{Types} & \TST &\bnfeq&
      \texttt{$\alpha$} \bnfor \texttt{\TST|\TST} 
      \\ &&\bnfor& \texttt{any()} \bnfor \texttt{none()}
      \\ &&\bnfor& \texttt{atom()} \bnfor \texttt{:$k$}
      \\ &&\bnfor& \texttt{pid()} \bnfor \texttt{port()} \bnfor \texttt{reference()}
      \\ &&\bnfor& \texttt{float()} \bnfor \texttt{integer()} 
      \\ &&\bnfor& \texttt{neg\_integer()} \bnfor \texttt{non\_neg\_integer()} \bnfor \texttt{pos\_integer()}
      \\ &&\bnfor& \texttt{tuple()} \bnfor \tuple{\ov{\TST}} \bnfor \map{\ov{\TSF}} 
      \\ &&\bnfor& \texttt{[]} \bnfor \texttt{nonempty\_maybe\_improper\_list(\TST, \TST)} 
      \\ &&\bnfor& \texttt{<<>>} \bnfor \texttt{<<\_::$n$>>} \bnfor \texttt{<<\_::\_*$n$>>} \bnfor \texttt{<<\_::$n$, \_::\_*$n$>>}
      \\ &&\bnfor& \texttt{(\fun{$\ov{\TST}$}{\TST})} \bnfor \texttt{(\fun{...}{\TST})}
      \\ &&\bnfor& \texttt{$n$} \bnfor \text{$n$..$n'$}
      \\
      \textbf{Fields} & \TSF &\bnfeq&
      \texttt{required(\TST)\,=>\,\TST} \bnfor \texttt{optional(\TST)\,=>\,\TST} 
      \end{grammarfig}
   \small(where $x$ ranges over expression variables, $\alpha$ ranges over type variables, 
   $n$ ranges over integers with $n<n'$, and $k$ ranges over atom names)
   \caption{TypeSpec}
   \label{app:typespec-syntax}
\end{figure}

\section{Translation from TypeSpec to Elixir Type}

\[
 \dsb{\,\,} \texttt{: TypeSpec $\rightarrow$ Elixir type}
\]

\subsection{Expressions}

% TypeSpec and Elixir type are structurally different while they share semantic similarities

% A function annotation in TypeSpec mainly consists of "@spec", input type, and output type. 
% Starting with "@spec" that announces the existence of an annotation, 
% the following expression is divided into input and output types by "::".
% Where input type has a function's name prepending that it specifies and places in between a pair of parentheses.
% In addition, TypeSpec allows to use type parameters by entailing a "when" clause which taken as a type in the whole clause.

% Elixir type features the semantical similarities described above 
% that are "@spec" is replaced by "\$" sign, and input and output types are only distinquished by "->".

% Notwithstanding the similarities, the main difference comes from TypeSpec's support in multi-clause annotations.
% It 

% It can consist of one or more clauses to express the polymorphic nature of Elixir functions.
% (Does order matter in Dialyzer??? Let's check!!!)

% while that in Elixir Type takes a single clause.
% In the case of TypeSpec's multi-clause, it can be represented with `and` between clauses in Elixir Type within a single clause.
% That is because multi-clauses indicates the existence of the matching function.
% Some may argue that the `or` can be an alternative. 
% However, it is not regarded since the former provides higher precision necessitating the presence of functions matching the annotations

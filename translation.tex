\section{Basic Types}

\begin{enumerate}

   \item Top types
   \begin{itemize}
      \item any() == var == term() \\
      => Compatible in the type migration as Elixir Type has term()
   \end{itemize}

   \item Bottom type
   \begin{itemize}
      \item none() \\
      => Compatible in the type migration
   \end{itemize}

   \item Identifier types
   \begin{itemize}
      \item pid()\\
      A pid (process identifier) refers into a process table and a node table, which also consumes memory. \\
      1 word for a pid from the current local node.
      
      \item port()\\
      A port identifier refers into a port table and a node table, which also consumes memory.
      
      \item reference()\\
      A reference also refers into more or less emulator internal data structures which also consumes memory.
      At a minimum it refers into the node tables.      
      \\
      => All compatible in the type migration
   
   \end{itemize}

   \item Atom types
   \begin{itemize}
      \item atom()
      An atom, ‘$k$’, can be used as a type while ‘true’, ‘false’, ‘nil’ are reserved atoms that are used without prepositional ':'.
      \\
      => Compatible in the type migration
   \end{itemize}

   \item Number types
   \begin{itemize}
      \item float()\\
      => Compatible in the type migration
   
      \item integer()\\
      TypeSpec's integer() has subtypes that are neg\_integer(), non\_neg\_integer(), and pos\_integer().
      Furthermore, an individual integer, ‘$n$’, can be used as a type while it supports range of integer with ‘..’ operator.
      Whereas, in Elixir type, integer() is indivisible.
      \\
      However, the future work could either split integer() into:
         individual elements;
         ranges;
         or neg\_integer() | zero() | pos\_integer().
      \\
      => Subtypes of integer() are incompatible in the type migration
   \end{itemize}

   \item Map types
   \begin{itemize}
      \item map()\\
      \begin{itemize}
         \item TypeSpec Syntactic Sugars\\
         \begin{enumerate}
            \item Empty map: \\
            \%\{\}
            \\
            \item Map with required key ‘$:key$’ of value\_type: \\
            \%\{key: value\_type\} == \%\{required(atom()) => \TST\}
            \\
            \item Map with required pairs of key\_type and value\_type: \\
            \%\{key\_type => value\_type\}
            \\ == \%\{required(key\_type) => value\_type\}
            \\ == \%\{required(\TST) => \TST\}
            \\
            \item Map with optional pairs of key\_type and value\_type \\
            \%\{optional(key\_type) => value\_type\}
            \\ == \%\{optional(\TST) => \TST\}
            \\
            \item Struct with all fields of any type \\
            \%SomeStruct\{\} == \%\{required(:\_\_struct\_\_) => $x$\}
            \\
            \item Struct with required key :key of value\_type \\
            \%SomeStruct\{key: value\_type\}
            \\ == \%\{required(:\_\_struct\_\_) => $x$\, required($k$) => \TST\}
            \\
         \end{enumerate}
         => Compatible in the type migration \\
         %\item New in Elixir type
         %\%\{\map{...\,\ov{\ETF}}\}
      \end{itemize}
   \end{itemize}
   
   \item List types
   \begin{itemize}
      \item Lists of types \\
      list(type) ==> empty\_list() or non\_empty\_list(\ETT)
      \\
      \item Non-empty lists of types \\
      non\_empty\_list(type)
      \\ ==> non\_empty\_list(\ETT)
      \\ == non\_empty\_list(\ETT, empty\_list())
      \\
      \item Proper or improper lists \\
      maybe\_improper\_list(content\_type, termination\_type)
      \\ ==> list(\ETT) or non\_empty\_list(\ETT, \ETT\,and not list())
      \\
      \item Improper lists \\
      nonempty\_improper\_list(content\_type, termination\_type)
      \\ ==> non\_empty\_list(\ETT, \ETT\,and not list())
      \\
      \item Non-empty proper or improper lists \\
      nonempty\_maybe\_improper\_list(content\_type, termination\_type)
      \\ ==> non\_empty\_list(\ETT) or non\_empty\_list(\ETT, \ETT\,and not list())
   \end{itemize}

\end{enumerate}

\section{Formalized TypeSpec and Elixir Type}

\begin{figure} [H]
   \begin{grammarfig}
      \textbf{Annotations} & \FWA &\bnfeq& 
      \TSAn \bnfor \varepsilon
      \\
      \textbf{Expressions} & \TSE &\bnfeq& 
      \TSFn{\TST} \bnfor 
      \TSFn{\TST}\,\,\texttt{when}\,\,\ov{\texttt{$\alpha$:}\,\TST}
      \\[2mm]
      \textbf{Types} & \TST &\bnfeq&
      \texttt{$\alpha$} \bnfor \texttt{\TST|\TST} 
      \\ &&\bnfor& \texttt{any()} \bnfor \texttt{none()}
      \\ &&\bnfor& \texttt{atom()} \bnfor \texttt{:$k$}
      \\ &&\bnfor& \texttt{pid()} \bnfor \texttt{port()} \bnfor \texttt{reference()}
      \\ &&\bnfor& \texttt{float()} \bnfor \texttt{integer()} 
      \\ &&\bnfor& \textcolor{red}{\texttt{neg\_integer()} \bnfor \texttt{non\_neg\_integer()} \bnfor \texttt{pos\_integer()}}
      \\ &&\bnfor& \texttt{tuple()} \bnfor \tuple{\ov{\TST}} \bnfor \map{\ov{\TSF}} 
      \\ &&\bnfor& \texttt{[]} \bnfor \texttt{nonempty\_maybe\_improper\_list(\TST, \TST)} 
      \\ &&\bnfor& \textcolor{red}{\texttt{<<>>} \bnfor \texttt{<<\_::$n$>>} \bnfor \texttt{<<\_::\_*$n$>>} \bnfor \texttt{<<\_::$n$, \_::\_*$n$>>}}
      \\ &&\bnfor& \texttt{($\ov{\TST}$\,->\,\TST)} \bnfor \textcolor{red}{\texttt{(...\,->\,\TST)}}
      \\ &&\bnfor& \textcolor{red}{\texttt{$n$} \bnfor \text{$n$..$n'$}}
      \\
      \textbf{Fields} & \TSF &\bnfeq&
      \texttt{required(\textcolor{red}{\TST})\,=>\,\TST} \bnfor \texttt{optional(\textcolor{red}{\TST})\,=>\,\TST} 
      \end{grammarfig}
   \small(where $x$ ranges over expression variables, $\alpha$ ranges over type variables, 
   $n$ ranges over integers with $n<n'$, and $k$ ranges over atom names)
   \caption{TypeSpec}
   \label{app:typespec-syntax}
\end{figure}


\begin{figure} [H]
   \begin{grammarfig}
      \textbf{Annotations} & \ETA &\bnfeq& 
      \texttt{\$}\,\ETS \bnfor \texttt{\$}\,\ETS\,\,\texttt{when}\,\,\ov{\texttt{$\alpha$:}\,\ETT}
      \\
      \textbf{Signatures} & \ETS &\bnfeq& 
      \ETFn{\ETT} \bnfor \texttt{\ETS\,and\,\ETS}
      \\[2mm]
      \textbf{Types} & \ETT &\bnfeq&
      \texttt{$\alpha$} \bnfor \text{$n$..$n$} \bnfor \texttt{\ETT\,or\,\ETT}\bnfor \texttt{\ETT\,and\,\ETT}\bnfor \texttt{not\,\ETT}
      \\ &&\bnfor& \ETK \bnfor \texttt{term()} \bnfor \texttt{none()} \bnfor \texttt{dynamic()}
      \\ &&\bnfor& \tuple{\ov{\ETT}} \bnfor \tuple{\ov{\ETT},...} \bnfor \map{\ov{\ETF}} \bnfor \map{...,\ov{\ETF}}
      \\ &&\bnfor& \texttt{($\ov{\ETT}$\,->\,\ETT)}
      \\
      \textbf{Key types} & \ETK &\bnfeq&
      \texttt{atom()} \bnfor \texttt{:$k$}
      \\ &&\bnfor& \texttt{pid()} \bnfor \texttt{port()} \bnfor \texttt{reference()}
      \\ &&\bnfor& \texttt{float()} \bnfor \texttt{integer()} \bnfor \texttt{binary()}
      \\ &&\bnfor& \texttt{tuple()} \bnfor \texttt{open\_map()} \bnfor \texttt{fun()}
      \\ &&\bnfor& \texttt{empty\_list()} \bnfor \texttt{non\_empty\_list(\ETT, \ETT)} % for now...
      \\
      \textbf{Fields} & \ETF &\bnfeq&
      \texttt{:$k$\,=>\,\ETT} \bnfor \texttt{\ETK\,=>\,if\_set(\ETT)}
   \end{grammarfig}
   \small(where $x$ ranges over expression variables, $\alpha$ ranges over type variables, 
   $n..n$ ranges over integers while invisible to the users, and $k$ ranges over atom names)
   \caption{Elixir Type}
   \label{app:elixir-type-syntax}
\end{figure}
